    Project Introduction

    สวัสดีครับ ยินดีต้อนรับเข้าสู่โปรเจกต์ไททานิค (Titanic) ซึ่งเป็นเหตุการณ์ที่น่าสลดที่เรืออัปปางในวันที่ 15 เมษายน ค.ศ. 1912 ขอแสดงความเสียใจด้วยครับ แน่นอนว่า

    1. 🔥 K-Nearest Neighbors

    3. 🔥 Ridge/Lasso Regression
    4. 🔥 Decision Tree
    5. 🔥 Random Forest

    2. Travel_Style : รูปแบบการเดินทาง เช่น เดินทางคนเดียว, เดินทางกับพ่อแม่ลูก, เดินทางกับญาติพี่น้อง, หรือเดินทางเป็นแบบครอบครัว อาจบอกได้ว่ามีโอกาสรอดมากน้อยแค่ไหน

    รวมถึงผมจะเปลี่ยน 4 คอลัมน์ต่อไปนี้เป็น factor ก่อนเทรนโมเดล เพราะเราเห็นเป็นหมวดหมู่ชัดเจน ไม่เหมือนชื่อผู้โดยสารที่ไม่มีรูปแบบที่ชัดเจน


    1. Survived ประกอบด้วย 2 ค่า Not_Survived, Survived เลเวล 0,1 ตามลำดับ เลเวล 1 คือสิ่งที่โมเดลจะพยายามทำนาย
    2. Sex ประกอบด้วย 2 ค่า Male, Female เลเวล "male", "female" ตามลำดับ
    3. Embarked ประกอบด้วย 3 ค่า C, Q, S เลเวล "C", "Q", "S" ตามลำดับ
    4. Tarvel_Stlye ประกอบด้วย 4 ค่า Alone, With Parents/Children Only, With Siblings/Spouse Only, With Mixed Family เลเวลตามลำดับ

    Sex = factor(Sex, levels = c("male", "female")), # เปลี่ยนเป็น factor คอลัมน์ Sex

# สร้างฟังก์ชัน split data เป็น 2 ชุด train_data, test_data
split_data <- function(df = data_model) { # ฟังก์ชันนี้รับ data = data_model เป็นค่าเริ่มต้น
  set.seed(42) # ป้องกันการสุ่มซ้ำ
  n <- nrow(df) # นับจำนวนแถวเตรียม subset
  id <- sample(n, 0.8 * n) # สุ่มจำนวนแถวมา 80% ของจำนวนแถวทั้งหมด เก็บค่าไว้ที่ id
  train_data <- df[id, ] # แบ่งข้อมูลอยู่ใน train_data โดยใช้ id ที่สุ่มมาเมื่อสักครู่ 80% ของจำนวนข้อมูลทั้งหมด
  test_data <- df[-id, ] # แบ่งข้อมูล test_data โดยใช้ id อีก 20% ที่เหลือจากการแบ่งเมื่อสักครู่
  return( # เวลาเรียกใช้ split_data()
    list( # เรียก list ของชุดข้อมูลต่อไปนี้
      train = train_data,  # list ชื่อ train ที่เรากำหนดไว้ตอน subset
      test = test_data # list ชื่อ test ที่เราเรากำหนดไว้ตอนเหลือจาก subset
    )
  )
}

    7.9 🔥🔥 Comparative Analysis of Classification Models

    สร้างตาราง ประเมิน Classification Model ทั้งหมด 5 Models

    Code
    R 
    
# สร้างตาราง Evaluate 5 Models ด้วย Error
# ชื่อแต่ละโมเดล
model_name <- c("K-Nearest Neighbors", "Logitic Regression", 
                "Ridge/Lasso Regression", "Decision Tree", 
                "Random Forest")

# รวม Error ของ Train Data อยู่ในเวกเตอร์เดียวกัน 
train_err_model <- c(
  knn_train_err, logit_train_err, 
  glmnet_train_err, rpart_train_err, 
  rf_train_err)

# รวม Error ของ Test Data อยู่ในเวกเตอร์เดียวกัน
test_err_model <- c(
  knn_test_err, logit_test_err, 
  glmnet_test_err, rpart_test_err, 
  rf_test_err)

# ตาราง Evaluate โมเดล
fitting <- data_frame(
  model_name, 
  train_err_model,
  test_err_model
)

# transfrom คอลัมน์ เพิ่มคอลัมน์ Diff, Error_Change, Evaluate, Time_Train
fitting["Diff"] <- test_err_model - train_err_model # สร้างคอลัมน์ Diff คำนวณผลต่าง Error ระหว่าง Train และ Test
fitting["Error_Change"] <- if_else(fitting$Diff > 0, "Increased", # สร้างคอลัมน์ Error_Change บอกว่า Error เพิ่มขึ้นหรือลดลง
                                   "Decreased")

# สร้างคอลัมน์ Evaluate ตามเงื่อนไขว่าโมเดลเรานั้นเมื่อมาปรับใช้ปรับ unseen data แล้ว Performance ต่างกันมากไหม เช่น GoodFitting, WeakFittng, Overfitting
fitting["Evaluate"] <- if_else(
  fitting$Diff < 1, "GoodFitting", 
    if_else(fitting$Diff < 2, "WeakFittng", "Overfitting"))

# สร้าง list รวมเวลาเทรนของโมเดลทั้งหมด 5 โมเดล
time_model <- list(
  train_knn_time,
  train_logit_time, 
  train_glmnet_time, 
  train_rpart_time, 
  train_rf_time)

# แปลงเป็นตาราง พร้อม transpose ตาราง
time_model <- time_model |>
  data.frame() |> # ทำเป็นตารง
  t() # transpose

# ปรับ Format ทศนิยมคอลัมน์ตัวเลขให้มีระดับนัยสำคัญที่ 4 ตำแหน่ง
fitting$Diff <- fitting$Diff |> round(4)
fitting$train_err_model <- fitting$train_err_model |> round(4)
fitting$test_err_model <- fitting$test_err_model |> round(4)

# สร้างตารางสุดท้าย โดยเอาเวลาเทรนมาเข้าร่วมด้วย พร้อมเปลี่ยนชื่อคอลัมน์ให้เหมาะสม + เลือกเฉพาะคอลัมน์ที่จะแสดงผล
fitting <- fitting |> # นำตารางหลักมาแล้วเขียนทับตารางเดิม
  bind_cols(time_model) |> # นำคอลัมน์เวลาเวลาเทรนมารวมกับตารางหลัก
  # เลือกคอลัมน์ที่เราจะให้แสดงผล ปล. เราเอา elapsed เพราะนับเวลาจริงที่เราเทรนโมเดล จนได้โมเดลมา
  select(
    model_name, elapsed, train_err_model, 
    test_err_model, Diff, Error_Change, Evaluate) |>
  # เปลี่ยนชื่อคอลัมน์
  rename(
    Time_Train = elapsed, # เปลี่ยนชื่อคอลัมน์ เวลาเทรนโมเดล
    Model_Name = model_name, # เปลี่ยนชื่อคอลัมน์ ชื่อโมเดล
