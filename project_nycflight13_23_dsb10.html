1. Project Introduction

โปรเจกต์นี้จะช่วยให้คุณ make decision better เกี่ยวกับเที่ยวบินใน New York City ปี 2013 และ 2023

2. Analyze & Transform Data 📍

เริ่มต้นด้วย การสำรวจเพื่อการวิเคราะห์ข้อมูล หรือ EDA โดยผมมีข้อมูลเที่ยวบิน New York City ในปี 2013 และ 2023 สิ่งที่เราต้องการคือ

  1. ผมต้องการทราบว่า คนบินไปรัฐไหนบ้าง จำนวนเท่าไร เพื่อที่จะประกอบการตัดสินใจจะไปเที่ยว หรือทำกิจกรรมต่างๆกับครอบครัว (ไม่มีชื่อรัฐในฐานข้อมูล บอกว่าสนามบินปลายทางตั้งอยู่รัฐไหน)
  2. เมื่อได้รัฐที่อยากไปแล้ว ต่อมาต้องไปช่วงเดือนไหนดี จะได้ลาพักร้อนล่วงหน้าได้ จัดวางแผนเรื่องเวลาการเดินทาง
  3. เมื่อได้รัฐที่จะไป + ไปช่วงเดือนไหน ต่อมา สายการบินไหนล่ะดีสุด (สมมติว่าสายการบินที่ดีที่สุดสำหรับเราคือ เงื่อนไขต่อไปนี้)
    • ขาออกไฟต์ดีเลย์น้อย + ถึงก่อนเวลา ต่อเที่ยวบิน
    • เวลาบินน้อยสุด ต่อเที่ยวบิน
    • เราต้องมาถึงสนามบินอย่างน้อย 3 ชม.
      (เผื่อเวลา check in, ทานข้าว, ฉุกเฉิน) ซึ่งบางเที่ยวบินออกก่อนเวลา 2 ชม.
  4. เลือกไฟต์บิน กี่โมงดีที่ช่วยลดโอกาสเจอฝนตก และสนามบินที่ดีที่สุดระหว่าง EWR(NJ), JFK(NY), LGA(NY)

2.1 People flying to which states and how many?

2.1.1 Alert NA (Join)

ในหัวข้อนี้ คอลัมน์ ad1name (รัฐ) ถูกสร้างขึ้นมาจาก laditude, longitude ของสนามบินปลายทาง ที่อยู่ในตาราง aiports และเราจะนำ flights มารวมกับ airports โดยการใช้ left_join กำหนดให้ flights เป็นตารางหลัก และ airports เป็นตารางรอง ปรากฏว่า มี NA เกิดขึ้น หลังจากการ Join

สามารถดูรายละเอียด Code ได้ที่ด้านล่าง (Code ยาวนิดนึงนะครับ) ⬇︎ และภาพตารางผลลัพธ์ได้ครับ

Code
R 

cities <- ne_download(scale = "large", type = "populated_places", returnclass = "sf")

list_of_airports_dfs <- list(
  airports13 = airports13,
  airports23 = airports23
)

process_airport_data <- function(df) {
  df |>
    select(faa, name, lat, lon) |>
    st_as_sf(coords = c("lon", "lat"), crs = 4326, remove = FALSE)
}

airport_coords_list <- map(list_of_airports_dfs, process_airport_data)

process_matched_cities <- function(df) {
  df |>
    st_join(cities, join = st_nearest_feature)
}

matched_cities_list <- map(airport_coords_list, process_matched_cities)

colnames(matched_cities_list$airports13) <- tolower(colnames(matched_cities_list$airports13))
colnames(matched_cities_list$airports13)[2] <- "airport_name"

colnames(matched_cities_list$airports23) <- tolower(colnames(matched_cities_list$airports23))
colnames(matched_cities_list$airports23)[2] <- "airport_name"

state13 <- matched_cities_list$airports13 |>
  select(faa, adm1name, airport_name)

state23 <- matched_cities_list$airports23 |>
  select(faa, adm1name, airport_name)

n_flight_des13 <- flights13 |>
  select(dest) |>
  group_by(dest) |>
  summarise(n_flights = n()) |>
  arrange(- n_flights)

n_flight_des23 <- flights23 |>
  select(dest) |>
  group_by(dest) |>
  summarise(n_flights = n()) |>
  arrange(- n_flights)

prep_n_flights13 <- n_flight_des13 |>
  left_join(state13, by = c("dest" = "faa")) |>
  select(airport_name, dest, adm1name, n_flights)

styled_note("พบ NA ตามตารางด้านล่าง : ⬇︎")
prep_n_flights13 |> filter(!complete.cases(prep_n_flights13))

prep_n_flights23 <- n_flight_des23 |>
  left_join(state23, by = c("dest" = "faa")) |>
  select(airport_name, dest, adm1name, n_flights)

styled_note("พบ NA ตามตารางด้านล่าง : ⬇︎ (สังเกตุได้ว่าปี 2023 มีการอัพเดทชื่อสนามบินเข้ามาให้แล้ว)")
prep_n_flights23 |> filter(!complete.cases(prep_n_flights23))

2.1.2 Cleaning Data

สำหรับตัวย่อสนามบิน 4 ที่ ที่มี NA ผมได้ทำการค้นหามาแล้วว่า ชื่อสนามบินแต่ละที่คืออะไร และตั้งอยู่รัฐไหน โดยอ้างอิงจากลิงก์่ต่อไปนี้

ตั้งอยู่ Puerto Rico

  • SJU ➤ Luis Muñoz Marín International Airport
  • BQN ➤ Rafael Hernández Airport
  • PSE ➤ Mercedita Airport

ตั้งอยู่ U.S. Virgin Islands

  • STT ➤ Cyril E. King Airport
Code
R 

prep_n_flights13 <- n_flight_des13 |>
  left_join(state13, by = c("dest" = "faa")) |>
  select(airport_name, dest, adm1name, n_flights) |>
  mutate(
    adm1name = case_when(
      is.na(adm1name) & dest == "STT" ~ "U.S. Virgin Islands", Virgin Islands เข้าไปแทน NA ในคอลัมน์ adm1name
      is.na(adm1name) & dest == "BQN" ~ "Puerto Rico", เข้าไปแทน NA ในคอลัมน์ adm1name
      is.na(adm1name) & dest == "SJU" ~ "Puerto Rico", เข้าไปแทน NA ในคอลัมน์ adm1name
      is.na(adm1name) & dest == "PSE" ~ "Puerto Rico", เข้าไปแทน NA ในคอลัมน์ adm1name
      TRUE ~ adm1name
    ),
    airport_name = case_when(
      is.na(airport_name) & dest == "STT" ~ "Cyril E. King Airport", ให้ใส่ชื่อสนามบิน Cyril E. King Airport เข้าไปแทน NA ในคอลัมน์ airport_name
      is.na(airport_name) & dest == "BQN" ~ "Rafael Hernández Airport", ให้ใส่ชื่อสนามบิน Rafael Hernández Airport เข้าไปแทน NA ในคอลัมน์ airport_name
      is.na(airport_name) & dest == "SJU" ~ "Luis Muñoz Marín International Airport", คือ SJU ให้ใส่ชื่อสนามบิน Luis Muñoz Marín International Airport เข้าไปแทน NA ในคอลัมน์ airport_name
      is.na(airport_name) & dest == "PSE" ~ "Mercedita Airport", ให้ใส่ชื่อสนามบิน Mercedita Airport เข้าไปแทน NA ในคอลัมน์ airport_name
      TRUE ~ airport_name
    )
  )

2.3 Which airline is the best?

ณ ตอนนี้ เรามีแผนที่จะไปพักร้อนที่ Florida ในเดือน March ต่อมาเราจะมาเลือกสายการบินตามเงื่อนไขที่เรากำหนด แน่นอนว่า เราต้องอยากได้สายการบินที่ออกตรงเวลาหรืออย่างน้อยก็ถึงจุดหมายปลายทางที่เร็วกว่ากำหนด ดังนั้น ผมได้จัดลำดับความสำคัญกับสิ่งที่ต้องการมากที่สุดน 3 ข้อหลักๆ ตามลำดับไว้ดังนี้

  1. ไฟต์ดีเลย์ น้อยขาออก + ถึงก่อนเวลา (best)
  2. ใช้เวลาบินน้อยสุด + ระยะเวลาบินต่อเที่ยวบิน ใน 1 ปี
  3. เราต้องมาถึงสนามบินอย่างน้อย 3 ชม.
    (เพื่อเวลา Check-in, ทานข้าว, เวลาฉุกเฉินเร่งด่วน เพราะ บางสายการบิน มีการออกก่อนเวลา 2 ชม กว่าๆ)
Code
R 

the_best_airlines_2013 <- flights13 |>
  left_join(airlines13, by = "carrier") |>
  left_join(state13, by = c("dest" = "faa")) |>
  filter(adm1name == "Florida") |>
  select(name, adm1name, dep_delay, arr_delay, air_time) |>
  select(-adm1name) |>
  mutate(time_delay = (dep_delay + arr_delay)) |>
  select(-dep_delay, -arr_delay) |>
  arrange(time_delay) |>
  group_by(name) |>
  summarise(time_delay = sum(time_delay),
            air_time = sum(air_time),
            n_flights = n()) |>
  mutate(time_delay_per_flight = round(time_delay / n_flights),
         air_time_per_flight = round(air_time / n_flights)
        ) |>
  select(name, time_delay_per_flight, air_time_per_flight) |>
  arrange(time_delay_per_flight, air_time_per_flight)
